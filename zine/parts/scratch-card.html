Scratch cardScratch card

Scratch off a foil layer with your cursor or finger to reveal a hidden code.

Why it works: The prize text is real text underneath (screen readers can read it), and “Reveal without scratching” gives keyboard users the same result.

Try it:
Receipt: axe clean · contrast 6.28:1 · 1 tab stop · RTL ✓ · long text ✓
  • PASS axe, WCAG 2.1 A and AA: no violations in light or dark (9 rules passed).
  • PASS Tightest text contrast 6.28:1, where 4.5:1 is needed (light).
  • PASS Still passes contrast with tokens fitted to a yellow brand colour.
  • PASS Right-to-left with Arabic sample text fits a 390px phone.
  • PASS Every string 40% longer still fits a 390px phone.
  • PASS Only short hover and focus transitions (under 300ms).

Tab order

  1. button "Reveal without scratching"

What assistive tech is given

- paragraph:
  - text: Your discount code
  - strong: TEAR-25
- button "Reveal without scratching"

Automated checks in CI (axe in Chromium, reduced motion on). They catch a lot, not everything: still try it with a screen reader.

See it in the zine npx -y tearsheet add scratch-card

/* tearsheet scratch-card vca084053 · https://qusaialjbali.github.io/tearsheet/zine/parts/scratch-card.html */
function setup(root) {
  const canvas = root.querySelector('.scratch-foil');
  const reveal = root.querySelector('.scratch-reveal');
  const ctx = canvas.getContext('2d');
  let drawing = false;
  let last = null;
  let strokes = 0;

  const paint = () => {
    const r = canvas.getBoundingClientRect();
    const dpr = window.devicePixelRatio || 1;
    canvas.width = r.width * dpr;
    canvas.height = r.height * dpr;
    ctx.scale(dpr, dpr);
    const g = ctx.createLinearGradient(0, 0, r.width, r.height);
    g.addColorStop(0, '#b8bcc6');
    g.addColorStop(0.5, '#e9ebf0');
    g.addColorStop(1, '#9ea3ad');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, r.width, r.height);
    ctx.fillStyle = 'rgba(0,0,0,0.45)';
    ctx.font = '700 15px system-ui, sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('SCRATCH HERE', r.width / 2, r.height / 2 + 5);
    ctx.globalCompositeOperation = 'destination-out';
    ctx.lineWidth = 36;
    ctx.lineCap = 'round';
  };
  const cleared = () => {
    const data = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
    let clear = 0;
    for (let i = 3; i < data.length; i += 64) if (data[i] === 0) clear++;
    return clear / (data.length / 64);
  };
  const point = (e) => {
    const r = canvas.getBoundingClientRect();
    return [e.clientX - r.left, e.clientY - r.top];
  };
  const onDown = (e) => {
    drawing = true;
    last = point(e);
    canvas.setPointerCapture(e.pointerId);
  };
  const onMove = (e) => {
    if (!drawing) return;
    const p = point(e);
    ctx.beginPath();
    ctx.moveTo(last[0], last[1]);
    ctx.lineTo(p[0], p[1]);
    ctx.stroke();
    last = p;
    if (++strokes % 12 === 0 && cleared() > 0.5) root.classList.add('is-done');
  };
  const onUp = () => (drawing = false);
  const onReveal = () => root.classList.add('is-done');

  paint();
  canvas.addEventListener('pointerdown', onDown);
  canvas.addEventListener('pointermove', onMove);
  canvas.addEventListener('pointerup', onUp);
  reveal.addEventListener('click', onReveal);
  return () => {
    canvas.removeEventListener('pointerdown', onDown);
    canvas.removeEventListener('pointermove', onMove);
    canvas.removeEventListener('pointerup', onUp);
    reveal.removeEventListener('click', onReveal);
  };
}

document.querySelectorAll(".scratch").forEach(setup);
