Ink trail padInk trail pad

Move across the pad and leave a trail of ink dots that spread and fade like wet riso ink.

Why it works: Purely decorative, so the canvas is hidden from assistive tech and nothing depends on it. With reduced motion the dots stay put instead of spreading.

Try it:
Receipt: axe clean · contrast 6.7:1 · 0 tab stops · RTL ✓ · long text ✓
  • PASS axe, WCAG 2.1 A and AA: no violations in light or dark (7 rules passed).
  • PASS Tightest text contrast 6.7:1, where 4.5:1 is needed (light).
  • PASS Still passes contrast with tokens fitted to a yellow brand colour.
  • PASS Right-to-left with Arabic sample text fits a 390px phone.
  • PASS Every string 40% longer still fits a 390px phone.
  • PASS Movement stops or calms down with reduced motion on.

Tab order: nothing here takes focus (it is content, not controls).

What assistive tech is given

- figure "Draw with your pointer or finger"

Automated checks in CI (axe in Chromium, reduced motion on). They catch a lot, not everything: still try it with a screen reader.

See it in the zine npx -y tearsheet add ink-trail

script.js
/* tearsheet ink-trail v3e0552b1 · https://qusaialjbali.github.io/tearsheet/zine/parts/ink-trail.html */
function setup(root) {
  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const canvas = root.querySelector('canvas');
  const ctx = canvas.getContext('2d');
  const INKS = ['#ff48b0', '#0078bf'];
  let dots = [], raf = 0, last = null;
  const size = () => {
    const r = canvas.getBoundingClientRect(), dpr = window.devicePixelRatio || 1;
    canvas.width = r.width * dpr; canvas.height = r.height * dpr;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  };
  const frame = () => {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.globalCompositeOperation = 'multiply';
    dots = dots.filter((d) => d.life > 0);
    dots.forEach((d) => {
      if (!reduce) { d.r += 0.25; d.life -= 0.012; }
      ctx.globalAlpha = Math.max(0, d.life) * 0.8;
      ctx.fillStyle = d.ink;
      ctx.beginPath(); ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2); ctx.fill();
    });
    ctx.globalAlpha = 1;
    raf = dots.length && !reduce ? requestAnimationFrame(frame) : 0;
  };
  const onMove = (e) => {
    const r = canvas.getBoundingClientRect();
    const p = { x: e.clientX - r.left, y: e.clientY - r.top };
    const dist = last ? Math.hypot(p.x - last.x, p.y - last.y) : 99;
    if (dist < 6) return;
    last = p;
    dots.push({ x: p.x, y: p.y, r: 3 + Math.random() * 4, life: 1, ink: INKS[dots.length % 2] });
    if (reduce) frame();
    else if (!raf) raf = requestAnimationFrame(frame);
  };
  const onLeave = () => (last = null);
  size();
  window.addEventListener('resize', size);
  canvas.addEventListener('pointermove', onMove);
  canvas.addEventListener('pointerleave', onLeave);
  return () => {
    cancelAnimationFrame(raf);
    window.removeEventListener('resize', size);
    canvas.removeEventListener('pointermove', onMove);
    canvas.removeEventListener('pointerleave', onLeave);
  };
}

document.querySelectorAll(".trail").forEach(setup);
