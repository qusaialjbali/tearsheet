Chat threadChat thread

Message bubbles with grouping, timestamps, a typing indicator and a composer that sends.

Why it works: The message list is a log (role="log") so new messages are announced. Sent messages align right and carry the same text colour contrast as received ones.

Try it:
Receipt: axe clean · contrast 6.28:1 · 2 tab stops · RTL ✓ · long text ✓
  • PASS axe, WCAG 2.1 A and AA: no violations in light or dark (23 rules passed).
  • PASS Tightest text contrast 6.28:1, where 4.5:1 is needed (light).
  • PASS Still passes contrast with tokens fitted to a yellow brand colour.
  • PASS Right-to-left with Arabic sample text fits a 390px phone.
  • PASS Every string 40% longer still fits a 390px phone.
  • PASS Movement stops or calms down with reduced motion on.

  1. textbox "Message":
  2. button "Send message"

/* tearsheet chat-thread v523d8c85 · https://qusaialjbali.github.io/tearsheet/zine/parts/chat-thread.html */
function setup(root) {
  const form = root.querySelector('.chat-send');
  const input = form.querySelector('input');
  const log = root.querySelector('.chat-log');
  const typing = root.querySelector('.typing');
  const onSubmit = (e) => {
    e.preventDefault();
    const text = input.value.trim();
    if (!text) return;
    const li = document.createElement('li');
    li.className = 'msg me';
    const p = document.createElement('p');
    p.textContent = text;
    const t = document.createElement('time');
    const now = new Date();
    t.textContent = now.toTimeString().slice(0, 5);
    t.dateTime = t.textContent;
    li.append(p, t);
    log.insertBefore(li, typing);
    input.value = '';
  };
  form.addEventListener('submit', onSubmit);
  return () => form.removeEventListener('submit', onSubmit);
}

document.querySelectorAll(".chat").forEach(setup);
