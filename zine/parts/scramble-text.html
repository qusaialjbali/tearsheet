Decoder textDecoder text

A headline that decrypts itself from noise, one letter locking in at a time.

Why it works: The real sentence sits in visually hidden text and the scrambled copy is aria-hidden, so screen readers never hear the noise. It plays on hover and on the button, never on its own loop.

Try it:
Receipt: axe clean · contrast 17.41:1 · 1 tab stop · RTL ✓ · long text ✓
  • PASS axe, WCAG 2.1 A and AA: no violations in light or dark (9 rules passed).
  • PASS Tightest text contrast 17.41:1, where 4.5:1 is needed (dark).
  • PASS Still passes contrast with tokens fitted to a yellow brand colour.
  • PASS Right-to-left with Arabic sample text fits a 390px phone.
  • PASS Every string 40% longer still fits a 390px phone.
  • PASS Movement stops or calms down with reduced motion on.

Tab order

  1. button "Decode again"

What assistive tech is given

Automated checks in CI (axe in Chromium, reduced motion on). They catch a lot, not everything: still try it with a screen reader.

script.js
/* tearsheet scramble-text v00fecc37 · https://qusaialjbali.github.io/tearsheet/zine/parts/scramble-text.html */
function setup(root) {
  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const GLYPHS = '!<>-_/[]{}=+*^?#ABCDEFGHJKLMNPQRSTUVWXYZ';
  const textEl = root.querySelector('.decode-text');
  const span = textEl.querySelector('.decode-show');
  const button = root.querySelector('.decode-btn');
  const text = span.textContent;
  let raf;

  const run = () => {
    if (reduce) return;
    cancelAnimationFrame(raf);
    let frame = 0;
    const tick = () => {
      const frag = document.createDocumentFragment();
      let done = true;
      [...text].forEach((ch, i) => {
        if (ch === ' ' || frame > i * 2 + 8) {
          frag.append(ch);
        } else {
          done = false;
          const noise = document.createElement('span');
          noise.className = 'decode-noise';
          noise.textContent = GLYPHS[Math.floor(Math.random() * GLYPHS.length)];
          frag.append(noise);
        }
      });
      span.replaceChildren(frag);
      frame++;
      if (!done) raf = requestAnimationFrame(tick);
    };
    tick();
  };

  textEl.addEventListener('pointerenter', run);
  button.addEventListener('click', run);
  run();
  return () => {
    cancelAnimationFrame(raf);
    textEl.removeEventListener('pointerenter', run);
    button.removeEventListener('click', run);
  };
}

document.querySelectorAll(".decode").forEach(setup);
